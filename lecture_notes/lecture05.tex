\section{Multilinear Extensions}
%----------------------------------------------------------------------

We begin by recalling the setup from previous lectures and developing the theory of
\emph{multilinear extensions} (MLEs).

\subsection{Setup and Definition}

\begin{tcolorbox}[colback=blue!5!white, colframe=blue!50!black, title=Setup]
Let $\bits^v$ denote the set of all $v$-bit strings (i.e.\ binary strings of length $v$).
Any function
\[
  f : \bits^v \to \Fp
\]
can be viewed as an element of $\Fp^{2^v}$, since it assigns a field element to
each of the $2^v$ binary strings of length $v$.
\end{tcolorbox}

\begin{definition}[Extension of a function]
A $v$-variate polynomial $g$ \emph{extends} $f : \bits^v \to \Fp$ if
\[
  \forall\, i \in \bits^v, \quad g(i) = f(i),
\]
where $g : \Fp^v \to \Fp$ (writing $g(i) = g(i_0, i_1, \ldots, i_{v-1})$,
and where $i = \sum_j 2^j i_j$ interprets $i$ as an integer via its binary digits).
\end{definition}

\begin{remark}
The notation $g(i)$ for $i \in \bits^v$ should be understood as evaluating the polynomial
$g : \Fp^v \to \Fp$ at the point $(i_0, i_1, \ldots, i_{v-1}) \in \bits^v
\subset \Fp^v$.
\end{remark}

\subsection{Multilinear Extension (MLE): Existence and Uniqueness}

\begin{tcolorbox}[colback=green!5!white, colframe=green!50!black, title=Key Theorem (MLE)]
\begin{theorem}[Existence and Uniqueness of the MLE]
\label{thm:mle}
Any function $f : \bits^v \to \Fp$ has a \emph{unique} multilinear extension
$\widetilde{f} : \Fp^v \to \Fp$.
\end{theorem}
\end{tcolorbox}

\begin{proof}
\textbf{Construction.} Define the \emph{multilinear Lagrange basis polynomial} for a point
$w \in \bits^v$ by
\[
  \chi_w(x_1, \ldots, x_v)
  \;:=\;
  \prod_{i=1}^{v} \bigl( x_i w_i + (1 - x_i)(1 - w_i) \bigr).
\]
Observe that $\chi_w$ evaluates to 1 if $x = w$ and 0 for all other $x \in \bits^v$:
\[
  \chi_w(x) :=
  \begin{cases}
    1 & \text{if } x = w, \\
    0 & \text{otherwise (i.e., } x \neq w\text{)}.
  \end{cases}
\]
Now define
\[
  \widetilde{f}(x_1, \ldots, x_v)
  \;:=\;
  \sum_{w \in \bits^v} f(w)\,\chi_w(x_1, \ldots, x_v).
\]

We verify three claims:

\medskip
\textbf{Claim 1: $\widetilde{f}$ extends $f$.}
For any $x \in \bits^v$, exactly one term in the sum is nonzero (the term $w = x$), so
$\widetilde{f}(x) = f(x) \cdot 1 = f(x)$.

\medskip
\textbf{Claim 2: $\widetilde{f}$ is multilinear.}
Each $\chi_w$ is a product of degree-1 factors, one per variable; the overall polynomial
is therefore multilinear (degree at most 1 in each variable).
Because $\widetilde{f}$ is a linear combination of multilinear polynomials, it is itself
multilinear.
(Note: $\chi_w$ is multilinear because it is itself a linear combination of multilinear
polynomials; $\widetilde{f}$ is a linear combination of multilinear polynomials.)

\medskip
\textbf{Claim 3: $\widetilde{f}$ is unique.}
Any two multilinear extensions agree on all $2^v$ points of $\bits^v$.
Their difference is a multilinear polynomial that vanishes on $\bits^v$.
Since each variable has degree at most 1, one can show (by an inductive Schwartz--Zippel
style argument) that the difference must be the zero polynomial.
Hence $\widetilde{f}$ is the unique multilinear extension.
\end{proof}

\subsection{Factored Form of the Lagrange Basis and Efficient Evaluation}

The key structural property of $\chi_w$ is that it \emph{factors} across the variables:
\[
  \chi_w(x_1, \ldots, x_v)
  \;=\;
  \prod_{i=1}^{v} \bigl( x_i w_i + (1-x_i)(1-w_i) \bigr).
\]
This can be broken apart using a ``half-and-half'' split.
Writing $v/2$ for the midpoint, we have
\[
  \chi_w(x_1, \ldots, x_v)
  \;=\;
  \chi_{w_1,\ldots,w_{v/2-1}}(x_1, \ldots, x_{v/2-1})
  \;\cdot\;
  \chi_{w_{v/2},\ldots,w_{v}}(x_{v/2}, \ldots, x_v).
\]
This factored structure---``breaking $\widetilde{f}$ into chunks''---is the basis for efficient
algorithms to evaluate $\widetilde{f}$ at arbitrary points in $\Fp^v$.

\begin{remark}[Interpretation of $\chi_w$]
The factor for variable $i$ is:
\[
  x_i w_i + (1-x_i)(1-w_i)
  \;=\;
  \begin{cases}
    x_i & \text{if } w_i = 1, \\
    1 - x_i & \text{if } w_i = 0.
  \end{cases}
\]
This is a linear ``selector'' that equals 1 when $x_i = w_i$ and 0 when $x_i = 1 - w_i$.
\end{remark}

\begin{remark}[Computational significance]
The MLE $\widetilde{f}$ is central to many ZKP constructions (e.g., the sumcheck protocol).
Given oracle access to evaluations of $\widetilde{f}$, a verifier can efficiently check
properties of $f$ over the Boolean hypercube $\bits^v$ without reading all $2^v$ values.
\end{remark}

\section{Relations, Languages, and the Class NP}
%----------------------------------------------------------------------

We now develop the computational complexity background needed for the theory of
zero-knowledge proofs.

\subsection{Languages and Relations}

\begin{tcolorbox}[colback=blue!5!white, colframe=blue!50!black, title=Languages]
\begin{definition}[Language]
A \emph{language} $L$ is a subset of $\bits^*$ (the set of all finite binary strings).
Equivalently, $L$ is the set of objects sharing some common property.
\end{definition}
\end{tcolorbox}

\begin{example}[Languages]
\hfill
\begin{itemize}
  \item $L_{\mathrm{3CNFSAT}}$: the set of all satisfiable 3-CNF formulas.
  \item $L_{\mathrm{3COL}}$: the set of all 3-colorable graphs.
  \item Membership: $x \in L$ means ``$x$ has the property''.
\end{itemize}
\end{example}

\begin{tcolorbox}[colback=blue!5!white, colframe=blue!50!black, title=Relations]
\begin{definition}[Relation]
A \emph{relation} $R$ is a subset of $\bits^* \times \bits^*$.
\end{definition}
\end{tcolorbox}

\paragraph{From language to relation via witnesses}
Given a language $L$, we associate a relation $R_L$ via the notion of a \emph{witness}:
\[
  x \in L \quad \iff \quad \exists\, w \text{ s.t. } (x, w) \in R_L.
\]
Here:
\begin{itemize}
  \item $x$ is the \emph{instance} (the ``statement'' to be verified).
  \item $w$ is the \emph{witness} (the ``proof'' or ``certificate'').
\end{itemize}

\begin{example}[3-CNF-SAT relation]
For $L_{\mathrm{3CNFSAT}}$, the instance $x$ is a 3-CNF formula $\varphi$ and the witness
$w$ is a satisfying assignment.
$(x, w) \in R_{L_{\mathrm{3CNFSAT}}}$ iff $w$ is a satisfying assignment for $\varphi$.
\end{example}

\begin{definition}[Verifier / Checker]
A \emph{verifier} (or \emph{checker}) $V$ for $R_L$ is an algorithm that:
\begin{itemize}
  \item Takes input $(x, w)$.
  \item Outputs $1$ if $(x, w) \in R_L$, and $0$ otherwise.
\end{itemize}
\end{definition}

\subsection{The Class NP}

\begin{tcolorbox}[colback=orange!10!white, colframe=orange!60!black, title=Definition of NP]
\begin{definition}[NP]
$\mathbf{NP}$ is the class of languages $L$ that have a \emph{deterministic polynomial-time}
verifier $V$ for the relation $R_L$.
That is, $L \in \mathbf{NP}$ iff there exists a polynomial $p$ and a deterministic
algorithm $V$ running in time $p(|x|)$ such that:
\[
  x \in L \quad \iff \quad \exists\, w : V(x, w) = 1.
\]
\end{definition}
\end{tcolorbox}

\begin{example}
For 3-CNF-SAT, the verifier $V_{\mathrm{3CNFSAT}}(x, w)$ checks whether the assignment $w$
satisfies the formula $x$.  This can be done in polynomial time, so
$L_{\mathrm{3CNFSAT}} \in \mathbf{NP}$.
\end{example}

\begin{remark}
Intuitively, $\mathbf{NP}$ is the class of problems where solutions (witnesses) can be
\emph{verified} efficiently, even if they cannot be \emph{found} efficiently.
\end{remark}

\subsection{An NP-Complete Language: Rank-One Constraint Systems (R1CS)}

\begin{tcolorbox}[colback=purple!5!white, colframe=purple!60!black,
  title=Rank-One Constraint Systems (R1CS)]
\textbf{Parameters:} A field $\F$, and integers $m, n \in \N$.

\medskip
\textbf{Definition.}
An R1CS instance is defined by three matrices $A, B, C \in \F^{m \times n}$.
A vector $z \in \F^n$ \emph{satisfies} the R1CS if:
\[
  (A \cdot z) \circ (B \cdot z) = C \cdot z,
\]
where $\circ$ denotes the \emph{Hadamard product} (element-wise multiplication of vectors).
\end{tcolorbox}

\begin{remark}[Hadamard product notation]
The operator $\circ$ denotes element-wise multiplication of vectors: if $u, v \in \F^m$,
then $(u \circ v)_i = u_i \cdot v_i$.  This is also called the ``Hadamard product''.
\end{remark}

\paragraph{Public and private parts of $z$}
The vector $z \in \F^n$ can be split into a \emph{public} part $x \in \F^\ell$
and a \emph{private} part $w \in \F^{n-\ell}$, so that $z = (x, w)$, in the sense
that $z$ satisfies the R1CS.

\begin{remark}[R1CS and circuit satisfiability]
R1CS is an NP-complete problem: it is general enough to encode any computation expressible
by an arithmetic circuit.
Each constraint $(A_i \cdot z)(B_i \cdot z) = C_i \cdot z$ encodes one multiplication gate
(rank-one constraint).
This is why R1CS is the standard way to express NP statements in modern ZKP systems.
\end{remark}

\section{Cryptography Background}
%----------------------------------------------------------------------

We now introduce the cryptographic foundations needed to define zero-knowledge proofs:
security definitions, adversaries, and hard problems.

\subsection{Security Definitions and Adversaries}

\begin{tcolorbox}[colback=blue!5!white, colframe=blue!50!black,
  title=What is a Security Definition?]
A \emph{security definition} is a formal statement about a testable ability of an adversary
to ``do something'' (e.g., break a protocol).
\end{tcolorbox}

Security definitions typically have the following components:

\begin{enumerate}[leftmargin=2em]
  \item \textbf{Setup (parameters):} Establish public parameters, group descriptions, etc.
  \item \textbf{Generate challenge instance:} Produce the challenge that the adversary must
        solve.
  \item \textbf{Run the adversary:} Execute the adversary algorithm $\mathcal{A}$ on the
        challenge; the adversary may also be given access to \emph{oracles}.
  \item \textbf{Success condition:} Check whether the adversary's output satisfies the
        winning condition.
\end{enumerate}

\begin{remark}[Efficient adversaries]
In cryptography, adversaries are required to be \emph{efficient} (i.e., probabilistic
polynomial-time, or PPT).  Security is meaningful only against such computationally
bounded adversaries.
\end{remark}

\subsection{Example Security Definition: The Discrete Logarithm Problem}

The \emph{discrete logarithm} of $A$ in a group $\G$ with generator $G$ is the
integer $x$ such that $xG = A$ (using additive notation for the group operation).

\begin{tcolorbox}[colback=gray!10!white, colframe=gray!60!black,
  title=Discrete Logarithm (DL) Security Game]
\textbf{Game} $\mathrm{DL}^{\G,G,p}(\mathcal{A})$:

\medskip
\begin{algorithmic}[1]
  \State $x \xleftarrow{\$} \Z_p$ \Comment{Sample a random scalar}
  \State $x' \xleftarrow{\$} \mathcal{A}(xG,\, \G,\, G,\, p)$
         \Comment{Adversary attempts to recover $x$ given $xG$}
  \State \Return $\llbracket x = x' \rrbracket$
         \Comment{1 if adversary succeeds, 0 otherwise}
\end{algorithmic}
\end{tcolorbox}

\begin{definition}[DL Advantage]
The \emph{DL advantage} of an adversary $\mathcal{A}$ is:
\[
  \advan_{\mathrm{DL}}(\mathcal{A})
  \;=\;
  \Pr\!\left[\mathrm{DL}^{\G,G,p}(\mathcal{A}) = 1\right].
\]
\end{definition}

\begin{definition}[DL-hardness]
The discrete logarithm problem is \emph{hard} in $\G$ if for all
non-uniform PPT adversaries $\mathcal{A}$:
\[
  \advan_{\mathrm{DL}}(\mathcal{A}) = \negl(n),
\]
where $n$ is the security parameter and $\negl(\cdot)$ denotes a negligible function.
\end{definition}

\begin{remark}[The DL game, annotated]
\hfill
\begin{itemize}
  \item The adversary receives $xG$ (i.e., a random group element), the group $\G$,
        the generator $G$, and the group order $p$.
  \item The adversary outputs a guess $x'$ for the discrete log.
  \item In the game, the superscript $\mathcal{O}$ on $\mathcal{A}$ (visible in the slide)
        indicates that in some variants the adversary is given oracle access to additional
        resources.
  \item ``DL hard in $\G$'' means: for all nuPPT $\mathcal{A}$,
        $\advan_{\mathrm{DL}}(\mathcal{A}) = \negl(n)$.
\end{itemize}
\end{remark}

\section{Interactive Proofs}
%----------------------------------------------------------------------

We turn to the central object of this course: \emph{interactive proofs}.

\subsection{Overview}

An \emph{interactive proof} (IP) is a protocol between two parties:
\begin{itemize}
  \item A \textbf{prover} $P$ (potentially computationally unbounded), who wishes to
        convince the verifier that some statement is true.
  \item A \textbf{verifier} $V$ (efficient / polynomial-time), who checks the prover's
        claims.
\end{itemize}

\begin{center}
\begin{tikzpicture}[
  node distance=3.5cm,
  box/.style={draw, rectangle, rounded corners=4pt, minimum width=2.2cm,
              minimum height=1.0cm, align=center, font=\bfseries},
  arr/.style={-{Stealth[length=7pt]}, thick}
]
  \node[box, fill=blue!15] (P) {Prover $P$\\ (unbounded)};
  \node[box, fill=green!15, right=of P] (V) {Verifier $V$\\ (PPT)};

  \draw[arr, bend left=20] (P) to node[above, font=\small]{message $m_1$} (V);
  \draw[arr, bend left=20] (V) to node[below, font=\small]{challenge $c_1$} (P);
  \draw[arr, bend left=20] (P) to node[above, font=\small]{message $m_2$} (V);
  \node[below=0.5cm of V, font=\small, align=center]{accepts or\\rejects};
\end{tikzpicture}
\end{center}

The parties exchange messages in rounds, and at the end the verifier either \emph{accepts}
or \emph{rejects}.

\subsection{Properties of Interactive Proofs}

A good interactive proof system for a relation $R$ (language $L$) satisfies two key
properties:

\begin{tcolorbox}[colback=green!5!white, colframe=green!60!black,
  title=Properties of Interactive Proofs]
\begin{definition}[Completeness]
If $x \in L$ (i.e., $(x, w) \in R$ for some witness $w$) and both $P$ and $V$ follow
the protocol honestly, then $V$ \emph{accepts} with high probability (ideally probability 1).
\[
  \Pr[V \text{ accepts} \mid x \in L,\; P \text{ honest}] \geq 1 - \epsilon_c,
\]
where $\epsilon_c$ is the \emph{completeness error} (ideally $0$).
\end{definition}

\begin{definition}[Soundness]
If $x \notin L$, then for \emph{any} (possibly cheating) prover strategy $P^*$, the
verifier $V$ rejects with high probability:
\[
  \Pr[V \text{ accepts} \mid x \notin L,\; P^* \text{ arbitrary}] \leq \epsilon_s,
\]
where $\epsilon_s$ is the \emph{soundness error} (must be small).
\end{definition}
\end{tcolorbox}

\begin{remark}[Zero-knowledge]
In the context of ZKPs, a third property---\emph{zero-knowledge}---is also required:
the verifier learns nothing beyond the truth of the statement $x \in L$.
This will be formalized in later lectures.
\end{remark}

\section{(If Time) Finishing Up Schwartz--Zippel}
%----------------------------------------------------------------------

This section was included in the slides as a time-permitting topic.
The \emph{Schwartz--Zippel lemma} was introduced in a prior lecture and will be
revisited here if time allows.

\begin{theorem}[Schwartz--Zippel Lemma]
Let $p(x_1, \ldots, x_n)$ be a nonzero multivariate polynomial of total degree $d$ over
a field $\F$.  Let $S \subseteq \F$ be a finite set.  Then:
\[
  \Pr_{r_1,\ldots,r_n \,\xleftarrow{\$}\, S}\!\left[p(r_1,\ldots,r_n) = 0\right]
  \;\leq\;
  \frac{d}{|S|}.
\]
\end{theorem}

\begin{remark}[Application to MLEs]
The Schwartz--Zippel lemma is used to establish the uniqueness of the MLE (Theorem
\ref{thm:mle}): if two multilinear polynomials agree on all of $\bits^v$, then their
difference is a multilinear polynomial vanishing on $\bits^v$.
Applying Schwartz--Zippel shows that, when evaluated at a random point over $\Fp$,
the probability of a nonzero multilinear polynomial vanishing is at most $v/p$, which is
negligible for $p$ exponentially large.
\end{remark}

\section*{Summary}

This lecture covered four main topics:

\begin{enumerate}[leftmargin=2em]
  \item \textbf{Multilinear Extensions (MLEs).}
        Any function $f : \bits^v \to \Fp$ has a unique multilinear extension
        $\widetilde{f} : \Fp^v \to \Fp$.  The MLE is given explicitly by
        the Lagrange interpolation formula
        $\widetilde{f}(x) = \sum_{w \in \bits^v} f(w)\,\chi_w(x)$,
        where $\chi_w(x) = \prod_{i=1}^v (x_i w_i + (1-x_i)(1-w_i))$.
        The factored structure of $\chi_w$ enables efficient evaluation.

  \item \textbf{Relations, Languages, and NP.}
        A language $L \subseteq \bits^*$ is associated with a relation $R_L$ via
        witnesses: $x \in L \iff \exists w : (x,w) \in R_L$.
        The class $\mathbf{NP}$ consists of languages with deterministic polynomial-time
        verifiers.
        R1CS (rank-one constraint systems), defined by the equation
        $(A \cdot z) \circ (B \cdot z) = C \cdot z$, is an NP-complete language and
        the standard way to represent NP statements in ZKP systems.

  \item \textbf{Cryptography Background.}
        Security definitions are formal statements about an adversary's ability to break
        a protocol, typically consisting of: setup, challenge generation, adversary
        execution (with oracle access), and a success condition.
        An example is the Discrete Logarithm (DL) problem, which is conjectured to be
        hard for PPT adversaries in suitable groups.

  \item \textbf{Interactive Proofs.}
        An interactive proof involves a (potentially unbounded) prover and a PPT verifier
        exchanging messages.
        The two core properties are \emph{completeness} (an honest prover always convinces
        an honest verifier) and \emph{soundness} (a cheating prover cannot convince the
        verifier of a false statement).
\end{enumerate}
