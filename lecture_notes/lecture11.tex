\section{The Random Oracle Model}
%=============================================================

\subsection{Motivation and Definition}

The \emph{random oracle model} (ROM) is a heuristic framework for
designing and analyzing cryptographic protocols introduced by
Bellare and Rogaway in 1993~\cite{BellareRogaway93}.

\begin{tcolorbox}[colback=blue!5, colframe=blue!40!black,
                  title={The Random Oracle Model (Bellare--Rogaway '93)}]
\begin{itemize}
  \item Every party has oracle access to a \emph{global random function}
        $H : \{0,1\}^* \to \{0,1\}^\ell$.
  \item All parties---prover $\prover$ \emph{and} verifier $\verifier$---query
        the \emph{same} function $H$.  Consistency is guaranteed: if two
        parties query $H$ on the same input they receive the same output.
  \item The random oracle is \emph{lazily sampled}: the challenger
        maintains a table and, whenever a fresh input $x$ is queried,
        it samples a uniformly random output $H(x)$ and records it.
        Subsequent queries on the same $x$ return the same value.
\end{itemize}
\end{tcolorbox}

\subsection{The Random Oracle Methodology}

Designing and deploying a scheme in the ROM proceeds in two (or
three) steps:

\begin{enumerate}
  \item \textbf{Design.} Design the scheme (protocol) with oracle access
        to the random oracle $H$.
  \item \textbf{Prove.} Prove security of the scheme \emph{in the random
        oracle model}: the security reduction may query the RO on behalf
        of the adversary and use those queries to break an underlying
        hard problem.
  \item \textbf{Instantiate.} Instantiate the random oracle $H$ with a
        \emph{cryptographic hash function} such as SHA-256 or SHA-3 and
        deploy.
\end{enumerate}

\begin{remark}[Caveat on instantiation]
  Step 3 is \emph{difficult to justify rigorously}.  Canetti, Goldreich,
  and Halevi (CGH '98)~\cite{CGH98} showed \emph{uninstantiability}:
  there exist schemes that are secure in the ROM but insecure under
  \emph{every} concrete hash function.  Related notions:
  \begin{itemize}
    \item \textit{Uninstantiability} (CGH '98): some RO-secure schemes
          cannot be securely instantiated with any real hash function.
    \item \textit{Indifferentiability}: a framework for analyzing whether
          a concrete hash construction (e.g.\ the sponge) behaves like a
          random oracle in a composition sense.
    \item \textit{Rothblum} (and related work): further results on when
          the RO methodology is or is not sound.
  \end{itemize}
  Despite these caveats, the ROM methodology is ubiquitous in practice
  because it produces efficient, clean protocols whose security proofs
  give concrete guidance on parameter sizes.
\end{remark}

\subsection{Security in the ROM}

When we say a scheme is secure ``in the ROM'' we mean:

\begin{itemize}
  \item The adversary receives oracle access to the same function $H$.
  \item The security experiment (game) gives the adversary a
        \emph{polynomial} number of $H$-queries.
  \item The reduction can \emph{program} $H$ (choose outputs for
        unqueried inputs) and can \emph{observe} all adversary queries to
        $H$ -- this is the key extra power granted to the simulator in the
        ROM.
\end{itemize}

\section{The Fiat-Shamir Transform}
%=============================================================

\subsection{High-Level Idea}

Recall that a \emph{$\Sigma$-protocol} (sigma protocol) is a
three-message public-coin interactive proof:
\begin{enumerate}
  \item Prover $\prover$ sends a \emph{commitment} $\alpha_1$ (first message).
  \item Verifier $\verifier$ sends a uniformly random \emph{challenge}
        $\rho$ (second message, the ``verifier randomness'').
  \item Prover $\prover$ sends a \emph{response} $\alpha_2$ (third message).
\end{enumerate}

The key observation is that the verifier's only role in step 2 is to
sample a uniformly random challenge.  In the ROM, the prover can
\emph{simulate} this by querying the random oracle:

\begin{tcolorbox}[colback=green!5, colframe=green!40!black,
                  title={Fiat-Shamir: Replace the Verifier with a Random Oracle}]
  The verifier is \textbf{public-coin}: its only message is a uniformly
  random challenge.  Therefore, replace that message by a hash:
  \[
    \rho \;\leftarrow\; H(\mathbb{x},\, \alpha_1)
  \]
  where $\mathbb{x}$ is the public input (statement) and $\alpha_1$ is
  the prover's first message (commitment).  The challenge is now
  \emph{determined} by the public data already on the ``wire'', so no
  further interaction is needed.
\end{tcolorbox}

\subsubsection*{Formal description of the Fiat-Shamir transform}

Let $(\mathsf{P}_{\mathrm{SP}}, \mathsf{V}_{\mathrm{SP}})$ be a
$\Sigma$-protocol for a relation $\mathcal{R}$ with
\begin{itemize}
  \item first prover message (commitment) $\alpha_1$,
  \item verifier challenge $\rho$ drawn uniformly from some challenge
        space $\mathcal{C}$,
  \item second prover message (response) $\alpha_2$.
\end{itemize}
Let $H : \{0,1\}^* \to \mathcal{C}$ be a random oracle.
The \textbf{Fiat-Shamir non-interactive proof system}
$(\mathsf{P}^{H}, \mathsf{V}^{H})$ is defined as follows.

\begin{algorithm}[H]
\caption{Fiat-Shamir Prover $\mathsf{P}^{H}(\mathbb{x},\mathbb{w})$}
\begin{algorithmic}[1]
  \State Run $\mathsf{P}_{\mathrm{SP}}(\mathbb{x},\mathbb{w})$ to
         obtain the first message $\alpha_1$ and internal state $\tau$.
  \State Query the random oracle:
         $\rho \leftarrow H(\mathbb{x}, \alpha_1)$
         \quad \textit{(query transcript up to verifier message)}
  \State Continue running $\mathsf{P}_{\mathrm{SP}}$ with challenge $\rho$
         to obtain response $\alpha_2$.
  \State \Return proof $\pi \coloneqq (\alpha_1, \alpha_2, \tau)$.
\end{algorithmic}
\end{algorithm}

\begin{remark}
  In practice the ``state'' $\tau$ that is packaged into the proof
  may simply be empty (the prover state needed to answer the challenge
  is entirely determined by the witness and the randomness used to
  produce $\alpha_1$).  Many presentations omit $\tau$ and write
  $\pi = (\alpha_1, \alpha_2)$.
\end{remark}

\begin{algorithm}[H]
\caption{Fiat-Shamir Verifier $\mathsf{V}^{H}(\mathbb{x},\pi)$}
\begin{algorithmic}[1]
  \State Parse $\pi$ as $(\alpha_1, \alpha_2, \tau)$.
  \State Re-derive the SP randomness:
         $\rho \leftarrow H(\mathbb{x}, \alpha_1)$.
  \State Check the $\Sigma$-protocol decision:
         \Return $\mathsf{V}_{\mathrm{SP}}(\mathbb{x}, \alpha_1, \rho, \alpha_2)$.
\end{algorithmic}
\end{algorithm}

\subsection{Why Does This Work?}

The Fiat-Shamir transform converts an \emph{interactive} public-coin
protocol into a \emph{non-interactive} proof (in the ROM) with the
following key properties:

\begin{itemize}
  \item \textbf{Completeness.}  An honest prover computes $\rho =
        H(\mathbb{x},\alpha_1)$ and constructs a valid $\Sigma$-protocol
        transcript; the verifier re-derives the same $\rho$ and accepts.
  \item \textbf{Soundness.}  A cheating prover cannot predict $\rho =
        H(\mathbb{x},\alpha_1)$ before querying $H$ (since $H$ is a
        random oracle).  The security proof exploits the fact that the
        reduction sees all prover queries to $H$ and can ``rewind'' or
        ``program'' the oracle.
  \item \textbf{Zero-knowledge.}  The simulation proceeds by programming
        the RO: the simulator chooses $\rho$ first, generates a simulated
        $(\alpha_1,\alpha_2)$ for that $\rho$, and then programs
        $H(\mathbb{x},\alpha_1) \coloneqq \rho$.
\end{itemize}

\section{Other Related Transforms}
%=============================================================

The Fiat-Shamir transform is the most widely used technique for
making interactive proofs non-interactive, but several related
transforms have been developed for various settings.

\begin{tcolorbox}[colback=yellow!5, colframe=orange!60!black,
                  title={Variants and Extensions}]
\begin{description}[leftmargin=2em]
  \item[Fischlin / Pass transform.]
        Achieves \emph{straight-line extraction}: the extractor does not
        need to rewind the prover to extract a witness.  This is important
        for concurrent security and for composability.

  \item[Unruh transform / QROM.]
        Fiat-Shamir is not automatically secure against \emph{quantum}
        adversaries in the \emph{quantum random oracle model} (QROM).
        The Unruh transform is a modification of Fiat-Shamir that
        achieves provable security in the QROM.

  \item[``Quantum'' ROM (QROM).]
        In the QROM, the adversary can query the random oracle in
        \emph{quantum superposition}: it queries a superposition of
        inputs and receives a superposition of outputs.  Proving
        security in the QROM requires new proof techniques
        (one-way-to-hiding lemma, compressed oracle technique, etc.).

  \item[``BCS'' transform.]
        A generalization of Fiat-Shamir to \emph{multi-round}
        (public-coin) interactive proofs, not just three-message
        $\Sigma$-protocols.  Converts any public-coin interactive
        proof into a non-interactive argument in the ROM.
\end{description}
\end{tcolorbox}

\section{Security of the Fiat-Shamir Transform}
%=============================================================

\subsection{Overview}

We now discuss security of the Fiat-Shamir transform in more detail.
The standard result is:

\begin{theorem}[Fiat-Shamir security, informal]
  Let $(\mathsf{P}_{\mathrm{SP}}, \mathsf{V}_{\mathrm{SP}})$ be a
  $\Sigma$-protocol that is \emph{complete}, \emph{special-sound}
  (or knowledge-sound), and \emph{honest-verifier zero-knowledge
  (HVZK)}.  Then the Fiat-Shamir transform
  $(\mathsf{P}^{H}, \mathsf{V}^{H})$ is a \emph{non-interactive
  zero-knowledge (NIZK) argument of knowledge} in the random oracle
  model.
\end{theorem}

The proof of this theorem proceeds in three parts, corresponding to
the three properties:

\begin{enumerate}
  \item \textbf{Completeness} is immediate from completeness of the
        $\Sigma$-protocol.
  \item \textbf{Knowledge soundness.} The extractor in the ROM has
        oracle access to all $H$-queries made by the (cheating) prover.
        The key technique is the \emph{forking lemma}: by running the
        prover twice with different random oracle responses to the same
        query $(\mathbb{x},\alpha_1)$, the extractor obtains two
        accepting transcripts $(\alpha_1,\rho,\alpha_2)$ and
        $(\alpha_1,\rho',\alpha_2')$ with $\rho \neq \rho'$, from
        which the witness can be extracted (by special soundness of the
        $\Sigma$-protocol).
  \item \textbf{Zero-knowledge.} The simulator programs the RO to
        make a simulated proof look identical to a real proof from the
        verifier's perspective.
\end{enumerate}

\subsection{``Weak'' Fiat-Shamir Attacks}

A crucial subtlety arises in practice.  There is a class of
\emph{``weak'' Fiat-Shamir attacks} that exploit incorrect
implementations of the transform.

\begin{tcolorbox}[colback=red!5, colframe=red!60!black,
                  title={``Weak'' Fiat-Shamir Attacks}]
  The term \textbf{``weak'' Fiat-Shamir} refers to a variant of the
  transform in which the hash (random oracle query) for the challenge
  does \emph{not} include all public data that it should.

  \medskip
  \textbf{Correct} (``strong'') Fiat-Shamir:
  \[
    \rho \;\leftarrow\; H(\underbrace{\mathbb{x}}_{\text{statement}},\;
                          \underbrace{\alpha_1}_{\text{commitment}})
  \]

  \medskip
  \textbf{Incorrect} (``weak'') Fiat-Shamir (a common mistake):
  \[
    \rho \;\leftarrow\; H(\alpha_1) \quad \text{(statement $\mathbb{x}$ omitted!)}
  \]

  \medskip
  When the statement $\mathbb{x}$ (or the commitment $\alpha_1$) is
  omitted from the hash input, an adversary can \emph{attack} the
  scheme:
  \begin{itemize}
    \item If $\mathbb{x}$ is omitted, a proof valid for one statement
          may be repurposed for a different statement.
    \item If $\alpha_1$ is omitted, the ``commitment'' step provides
          no binding guarantee; the prover can choose $\alpha_1$
          \emph{after} seeing $\rho$, breaking soundness.
  \end{itemize}

  Several real-world implementations of ZKP systems have been found to
  be vulnerable to weak Fiat-Shamir attacks; this is an active area of
  security engineering.
\end{tcolorbox}

\begin{remark}
  The correct rule is: the random oracle input must include
  \emph{all} public information that was available before the
  challenge was sampled in the interactive protocol.  In a
  $\Sigma$-protocol this means at minimum:
  $(\mathbb{x},\, \alpha_1)$.
  In a multi-round protocol it means the public input together with
  \emph{every} prover message sent before that round's challenge.
\end{remark}

\section{Example: Fiat-Shamir for Schnorr}
%=============================================================

\subsection{Recap: The Schnorr $\Sigma$-Protocol}

Let $\mathbb{G}$ be a cyclic group of prime order $p$ with generator
$G$.  The \emph{Schnorr identification protocol} is a $\Sigma$-protocol
for the discrete-logarithm relation
\[
  \mathcal{R}_{\mathrm{DL}} \;=\;
  \bigl\{\,((\mathbb{G},G,p,X),\; x) \;\bigm|\;
          X = xG\,\bigr\}.
\]
In the interactive version:
\begin{enumerate}
  \item \textbf{Prover} samples $r \xleftarrow{\$} \Z_p$, computes
        $R \leftarrow rG$, and sends $R$ to the verifier.
  \item \textbf{Verifier} samples $c \xleftarrow{\$} \Z_p$ and sends
        $c$ to the prover.
  \item \textbf{Prover} computes $z \leftarrow r + cx$ and sends $z$.
  \item \textbf{Verifier} accepts iff $zG = R + cX$.
\end{enumerate}

\subsection{Applying Fiat-Shamir to Schnorr}

Applying the Fiat-Shamir transform with random oracle
$F : \{0,1\}^* \to \Z_p$ produces the \emph{Schnorr signature / NIZK}
scheme.  The public input is $(\mathbb{G}, G, p, X)$.

\begin{tcolorbox}[colback=blue!5, colframe=blue!40!black,
                  title={Fiat-Shamir Schnorr: Non-Interactive Proof}]

\begin{minipage}[t]{0.46\textwidth}
\textbf{Prover} $P^F(\mathbb{G},G,p,X\,;\,x)$:
\medskip\hrule\medskip
\begin{algorithmic}[1]
  \State $r \xleftarrow{\$} \Z_p$;\quad $R \leftarrow rG$
  \State $c \leftarrow F(\mathbb{G},G,p,X,R)$
  \State $z \leftarrow r + cx$
  \State \Return $(R,\, c,\, z)$
\end{algorithmic}
\end{minipage}
\hfill
\begin{minipage}[t]{0.46\textwidth}
\textbf{Verifier} $V^F(\mathbb{G},G,p,X)$:
\medskip\hrule\medskip
\begin{algorithmic}[1]
  \State \textbf{Input:} proof $(R,\, c,\, z)$
  \State $c' \leftarrow F(\mathbb{G},G,p,X,R)$
  \State \textbf{If} $c \neq c'$: \Return $\bot$
  \State \Return $\bigl[\,zG = R + cX\,\bigr]$
\end{algorithmic}
\end{minipage}

\end{tcolorbox}

\subsection{Correctness Check}

Given an honestly generated proof $(R,c,z)$ with $z = r + cx$ and
$c = F(\mathbb{G},G,p,X,R)$:
\[
  zG \;=\; (r + cx)G \;=\; rG + c(xG) \;=\; R + cX. \quad\checkmark
\]
The verifier's check $zG = R + cX$ therefore passes.

\subsection{Security Intuition}

\begin{itemize}
  \item \textbf{Knowledge soundness.}  Suppose a cheating prover
        produces an accepting proof $(R,c,z)$ with $c = F(\mathbb{G},G,p,X,R)$.
        By the forking lemma, with non-negligible probability one can
        rewind the prover (or reprogram the RO) to obtain a second
        accepting proof $(R,c',z')$ with $c' \neq c$.  From the two
        equations
        \[
          zG = R + cX, \qquad z'G = R + c'X,
        \]
        we get $(z-z')G = (c-c')X$, so
        $x = (z-z')(c-c')^{-1} \pmod{p}$.
        This extracts the witness $x$, reducing knowledge soundness to
        the discrete logarithm problem.

  \item \textbf{Zero-knowledge.}  The simulator $\mathsf{Sim}$ (without
        witness) proceeds as follows: pick $c \xleftarrow{\$} \Z_p$ and
        $z \xleftarrow{\$} \Z_p$, set $R \leftarrow zG - cX$, and
        program $F(\mathbb{G},G,p,X,R) \coloneqq c$.  The simulated
        proof $(R,c,z)$ is statistically identical to a real proof.

  \item \textbf{When used as a digital signature.}  Including the
        message $m$ in the hash---$c \leftarrow
        F(\mathbb{G},G,p,X,R,m)$---yields the \emph{Schnorr signature
        scheme}, which is existentially unforgeable under chosen-message
        attack (EUF-CMA) in the ROM under the DL assumption.
\end{itemize}
